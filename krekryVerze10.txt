// ---------------------- LED ----------------------
void ledTask(void*) {
  uint32_t k = 0;
  for (;;) {
    k++;
    digitalWrite(LED_POWER, HIGH);
    if (obdLed == 3) digitalWrite(LED_OBD, (k / 5) % 2);
    else digitalWrite(LED_OBD, obdLed == 2 ? HIGH : LOW);
    digitalWrite(LED_TX, txLed ? (k % 2) : LOW);
    vTaskDelay(pdMS_TO_TICKS(100));
  }
}

// ---------------------- tlačítko stránek (GPIO 13) ----------------------
// krátký stisk (< 500 ms) = další stránka, dlouhý (> 2 s) = nastavení
void btnTask(void*) {
  bool stabilni = HIGH, posledni = HIGH, dlouhy = false;
  uint32_t zmena = 0, stisk = 0;
  for (;;) {
    uint32_t now = millis();
    bool r = digitalRead(BTN_PAGE);
    if (r != posledni) { posledni = r; zmena = now; }
    if (now - zmena > 30 && r != stabilni) {       // zákmity: stav musí držet 30 ms
      stabilni = r;
      if (stabilni == LOW) { stisk = now; dlouhy = false; }
      else if (!dlouhy && now - stisk < 500) {
        stranka = (stranka + 1) % 3;
        rucniMs = now ? now : 1;                   // ukázat zvolenou stránku i při stání / mimo auto
      }
    }
    if (stabilni == LOW && !dlouhy && now - stisk > 2000) {
      dlouhy = true;
      portalReq = true;
    }
    vTaskDelay(pdMS_TO_TICKS(10));
  }
}

void setObdSt(uint8_t st) {
  static uint8_t last = 255;
  if (st == last) return;
  last = st;
  if (prefs.getUChar("obdSt", 0) != st) prefs.putUChar("obdSt", st);
}

void setWifiSt(uint8_t st) {
  if (prefs.getUChar("wifiSt", 0) != st) prefs.putUChar("wifiSt", st);
}

String duvodText(uint8_t d) {
  if (d == 202) return "špatné heslo";
  if (d == 15 || d == 204) return "špatné heslo nebo slabý signál";
  if (d == 201) return "síť nenalezena";
  if (d == 200 || d == 203 || d == 2 || d == 4 || d == 8) return "slabý signál";
  if (d == 0) return "vypršel čas";
  return "kód " + String(d);
}

String signalText(int r) {
  if (!r) return "";
  String q = r >= -67 ? "dobrý" : r >= -80 ? "slabý" : "velmi slabý";
  return " · signál " + String(r) + " dBm (" + q + ")";
}

  uint8_t o = prefs.getUChar("obdSt", 0), w = prefs.getUChar("wifiSt", 0);
  String tg = prefs.getString("tgSt", "");
  int q = prefs.getBytesLength("queue") / sizeof(Trip);
  String h = "<div class='diag'>";
  h += "OBD: ";
  h += o == 3 ? "✅ spojeno" : o == 2 ? "⚠️ adaptér spojen, auto neodpovídá" :
       o == 1 ? "❌ adaptér nenalezen" : "– zatím nezkoušeno";
  h += "<br>Domácí Wi-Fi: ";
  if (w == 2) h += "✅ připojeno";
  else if (w == 1) h += "❌ nepřipojilo se – " + duvodText(prefs.getUChar("wifiDuv", 0));
  else h += "– zatím nezkoušeno";
  h += signalText((int8_t)prefs.getChar("rssi", 0));
  h += "<br>Telegram: ";
  h += !tg.length() ? String("– zatím nic neposláno") : tg == "OK" ? String("✅ OK") : "❌ " + htmlEsc(tg);
  h += "<br>Čeká na odeslání: " + String(q);
  h += "</div>";
  return h;
}

// ---------------------- levely ----------------------
int levelFor(float km) {
  int l = 1;
  for (int i = 1; i < 10; i++) if (km >= LVL_KM[i]) l = i + 1;
  return l;
}

const char* levelIcon(int l) {
  if (l <= 2) return "🟢";
  if (l <= 4) return "🔵";
  if (l <= 6) return "🟣";
  if (l <= 8) return "🟠";
  if (l == 9) return "🔴";
  return "⚫️";
}

float totalKm() { return prefs.getFloat("totalKm", 0); }

String jmenoDlouhe(float km) {
  int l = levelFor(km);
  String s = cfg.name + " " + levelIcon(l) + " Lvl " + String(l);
  if (cfg.car.length()) s += " · " + cfg.car;
  return s;
}

String jmenoKratke(float km) {
  int l = levelFor(km);
  String s = cfg.name + " " + levelIcon(l) + String(l);
  if (cfg.car.length()) s += " · " + cfg.car;
  return s;
}

// ---------------------- ukládání ----------------------
void saveTrip()  { prefs.putBytes("trip", &trip, sizeof(Trip)); }
void saveRec()   { prefs.putBytes("rec", &rec, sizeof(Records)); }
void saveQueue() {
  if (qCount) prefs.putBytes("queue", queue, qCount * sizeof(Trip));
  else prefs.remove("queue");
}

void loadAll() {
  if (prefs.getBytesLength("trip") == sizeof(Trip)) prefs.getBytes("trip", &trip, sizeof(Trip));
  else memset(&trip, 0, sizeof(Trip));

  if (prefs.getBytesLength("rec") == sizeof(Records)) prefs.getBytes("rec", &rec, sizeof(Records));
  else memset(&rec, 0, sizeof(Records));

  size_t qs = prefs.getBytesLength("queue");
  if (qs && qs % sizeof(Trip) == 0 && qs / sizeof(Trip) <= MAX_QUEUE) {
    qCount = qs / sizeof(Trip);
    prefs.getBytes("queue", queue, qs);
  } else qCount = 0;
}

// ---------------------- jízdy ----------------------
void startTrip() {
  memset(&trip, 0, sizeof(Trip));
  trip.open = 1;
  trip.maxCoolant = -100;
  saveTrip();
}

void finishTrip() {
  if (!trip.open) return;
  trip.open = 0;
  if (trip.distKm >= MIN_TRIP_KM) {
    float celkem = totalKm() + trip.distKm;
    prefs.putFloat("totalKm", celkem);
    trip.totalKm = celkem;

    trip.recFlags = 0;
    if (trip.best050 && (!rec.best050 || trip.best050 < rec.best050)) {
      trip.prev050 = rec.best050; rec.best050 = trip.best050; trip.recFlags |= 1;
    }
    if (trip.best0100 && (!rec.best0100 || trip.best0100 < rec.best0100)) {
      trip.prev0100 = rec.best0100; rec.best0100 = trip.best0100; trip.recFlags |= 2;
    }
    if (trip.maxSpeed > rec.maxSpeed) {
      trip.prevMaxSpeed = rec.maxSpeed; rec.maxSpeed = trip.maxSpeed; trip.recFlags |= 4;
    }
    saveRec();
    if (qCount == MAX_QUEUE) { memmove(queue, queue + 1, sizeof(Trip) * (MAX_QUEUE - 1)); qCount--; }
    queue[qCount++] = trip;
    saveQueue();
  }
  saveTrip();
}

// ---------------------- formátování ----------------------
String comma(String s) { s.replace(".", ","); return s; }
String fmtKm(float km)    { return comma(String(km, 1)); }
String fmtS(uint32_t ms)  { return comma(String(ms / 1000.0f, 1)) + " s"; }
String fmtV(uint16_t mv)  { return comma(String(mv / 1000.0f, 1)) + " V"; }
String fmtDur(uint32_t s) {
  if (s < 60) return String(s) + " s";
  uint32_t h = s / 3600, m = (s % 3600) / 60;
  return h ? String(h) + " h " + String(m) + " min" : String(m) + " min";
}

String fmtRpm(uint16_t r) {
  if (r < 1000) return String(r);
  String z = String(1000 + r % 1000);
  return String(r / 1000) + " " + z.substring(1);
}

const char* ODDEL = "━━━━━━━━━━\n";

// zpráva o jízdě (formát verze 4 s emoji)
String tripMsg(const Trip& t) {
  String m = "🚗 " + jmenoDlouhe(t.totalKm) + "\n";
  m += "⏱ za volantem " + fmtDur(t.movingMs / 1000) + " · 📏 " + fmtKm(t.distKm) + " km\n";
  m += ODDEL;

  // výkon
  m += "🏁 0–50: " + (t.best050 ? fmtS(t.best050) : String("–"));
  if (t.recFlags & 1) m += t.prev050 ? " 🏆 rekord (předtím " + fmtS(t.prev050) + ")" : String(" 🏆");
  m += "\n🏁 0–100: " + (t.best0100 ? fmtS(t.best0100) : String("–"));
  if (t.recFlags & 2) m += t.prev0100 ? " 🏆 rekord (předtím " + fmtS(t.prev0100) + ")" : String(" 🏆");
  m += "\n💨 " + String(t.maxSpeed) + " km/h";
  if ((t.recFlags & 4) && t.prevMaxSpeed) m += " 🏆";
  m += " · 🔧 " + fmtRpm(t.maxRpm) + " ot/min\n";
  m += ODDEL;

  // stav auta
  if (!t.obdOk) {
    m += "⚠️ Auto neodpovídalo – žádná data z OBD";
    return m;
  }
  String radek = "";
  if (t.maxCoolant > -50) {
    radek += "🌡 " + String(t.maxCoolant) + " °C";
    if (t.maxCoolant >= 110) radek += " ⚠️";
  }
  if (t.battLast) {
    if (radek.length()) radek += " · ";
    radek += "🔋 " + fmtV(t.battLast);
    if (t.battLast < 12000) radek += " ⚠️";
  }
  if (!t.mil && !t.dtc) {
    if (radek.length()) radek += " · ";
    radek += "✅ bez chyb";
  }
  m += radek;
  if (t.mil)       m += "\n⚠️ Svítí kontrolka motoru! Chyb: " + String(t.dtc);
  else if (t.dtc)  m += "\n⚠️ Uložené chyby motoru: " + String(t.dtc);
  return m;
}

// ---------------------- OLED displej ----------------------
String ascii(const String& s) {
  String o;
  for (size_t i = 0; i < s.length(); i++) {
    uint8_t c = (uint8_t)s[i];
    if (c < 128) o += (char)c;
  }
  return o;
}

void radek(int r, const String& s) {
  oled.setTextSize(1);
  String a = ascii(s);
  if (a.length() > 21) a = a.substring(0, 21);
  oled.setCursor(0, r * 8);
  oled.print(a);
}

// velký řádek: malý popisek vlevo, hodnota dvojnásobným písmem vpravo
void radekVelky(int y, const String& popis, const String& hodnota) {
  oled.setTextSize(1);
  oled.setCursor(0, y + 4);
  oled.print(ascii(popis));
  oled.setTextSize(2);
  oled.setCursor(40, y);
  oled.print(ascii(hodnota).substring(0, 7));
  oled.setTextSize(1);
}

void oledInit() {
  Wire.begin(OLED_SDA, OLED_SCL);
  oledOk = oled.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  if (oledOk) {
    Wire.setClock(400000);
    oled.clearDisplay();
    oled.display();
    Serial.println("OLED: ok");
  } else {
    Serial.println("OLED nenalezen – zkontroluj piny (SDA 5, SCL 4) a adresu 0x3C/0x3D");
  }
}

// Varování se aktivuje jen když auto reálně jede (prevSpeed > 5), aby nevyskakovalo při zapnutém klíčku
bool alarmAktivni(String& a, String& b) {
  if (liveCool > VAROVANI_COOL_C) {
    a = "Chladici " + String(liveCool) + " C";
    b = "Prehrivani motoru!";
    return true;
  }
  if (prevSpeed > 5 && trip.battLast && trip.battLast < VAROVANI_BAT_MV) {
    a = "Baterie " + fmtV(trip.battLast);
    b = "Nenabiji alternator";
    return true;
  }
  return false;
}

// jízdní obrazovky podle tlačítka (stranka 0, 1, 2)
void drawDrive(uint32_t now) {
  if (stranka == 0) {                       // zrychlení – velké časy přes celý displej
    uint32_t d050  = run050  ? run050  : trip.best050;
    uint32_t d0100 = run0100 ? run0100 : trip.best0100;
    String t050  = runActive ? (run050  ? fmtS(run050)  : fmtS(now - runStart)) : (d050  ? fmtS(d050)  : String("--"));
    String t0100 = runActive ? (run0100 ? fmtS(run0100) : fmtS(now - runStart)) : (d0100 ? fmtS(d0100) : String("--"));

    radek(0, "JIZDA " + fmtKm(trip.distKm) + " km");
    radek(1, "Max " + String(trip.maxSpeed) + " km/h");
    radekVelky(20, "0-50", t050);
    radekVelky(44, "0-100", t0100);
  } else if (stranka == 1) {                // turbo a motor
    float plnici = (liveMap - 100) / 100.0f;      // bar nad atmosférou
    String zn = plnici >= 0 ? "+" : "";
    radek(0, "TURBO A MOTOR");
    radek(2, "Plnici: " + zn + comma(String(plnici, 2)) + " bar");
    radek(3, "Zatizeni: " + String(liveLoad) + " %");
    radek(4, "Sani: " + String(liveIat) + " C");
    radek(5, "Otacky: " + String(liveRpm));
  } else {                                  // chlazení a baterie (bez "Motor nebezi")
    radek(0, "CHLAZENI A BATERIE");
    radek(2, "Chladici: " + String(liveCool) + " C");

String htmlEsc(String s) {
  s.replace("&", "&amp;");
  s.replace("<", "&lt;");
  s.replace(">", "&gt;");
  s.replace("\"", "&quot;");
  return s;
}

String cleanField(String s, int maxLen) {
  s.replace("\r", " ");
  s.replace("\n", " ");
  s.replace(" – ", " - ");
  s.trim();
  if ((int)s.length() > maxLen) {
    int n = maxLen;
    while (n > 0 && ((uint8_t)s[n] & 0xC0) == 0x80) n--;
    s = s.substring(0, n);
    s.trim();
  }
  return s;
}

bool homeVisible() {
  WiFi.mode(WIFI_STA);
  int n = WiFi.scanNetworks(false, false, false, 150);
  bool found = false;
  for (int i = 0; i < n; i++) {
    if (WiFi.SSID(i) != cfg.ssid) continue;
    if (!found || WiFi.RSSI(i) > homeRssi) {
      homeRssi = WiFi.RSSI(i);
      homeCh = WiFi.channel(i);
      memcpy(homeBssid, WiFi.BSSID(i), 6);
    }
    found = true;
  }
  WiFi.scanDelete();
  return found;
}

void onWifiOdpojeno(WiFiEvent_t e, WiFiEventInfo_t info) {
  wifiDuvod = info.wifi_sta_disconnected.reason;
}

bool wifiConnect() {
  static bool udalost = false;
  if (!udalost) {
    WiFi.onEvent(onWifiOdpojeno, ARDUINO_EVENT_WIFI_STA_DISCONNECTED);
    udalost = true;
  }
  if (homeRssi && prefs.getChar("rssi", 0) != homeRssi) prefs.putChar("rssi", homeRssi);
  for (int pokus = 0; pokus < 3; pokus++) {
    WiFi.disconnect(true);
    delay(200);
    wifiDuvod = 0;
    WiFi.mode(WIFI_STA);
    if (homeCh && pokus < 2) WiFi.begin(cfg.ssid.c_str(), cfg.pass.c_str(), homeCh, homeBssid);
    else WiFi.begin(cfg.ssid.c_str(), cfg.pass.c_str());
    uint32_t t = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - t < 15000) delay(100);
    if (WiFi.status() == WL_CONNECTED) return true;
    Serial.printf("WiFi pokus %d nevysel, duvod %d\n", pokus + 1, wifiDuvod);
    if (wifiDuvod == 202) break;
  }
  if (prefs.getUChar("wifiDuv", 0) != wifiDuvod) prefs.putUChar("wifiDuv", wifiDuvod);
  return false;
}

String urlencode(const String& s) {
  String o; char b[4];
  for (size_t i = 0; i < s.length(); i++) {
    uint8_t c = (uint8_t)s[i];
    if (isalnum(c) || c == '-' || c == '_' || c == '.' || c == '~') o += (char)c;
    else { snprintf(b, sizeof(b), "%%%02X", c); o += b; }
  }
  return o;
}

// ---------------------- JSON pomůcky ----------------------
void appendUtf8(String& o, uint32_t cp) {
  if (cp < 0x80) o += (char)cp;
  else if (cp < 0x800) { o += (char)(0xC0 | (cp >> 6)); o += (char)(0x80 | (cp & 0x3F)); }
  else if (cp < 0x10000) { o += (char)(0xE0 | (cp >> 12)); o += (char)(0x80 | ((cp >> 6) & 0x3F)); o += (char)(0x80 | (cp & 0x3F)); }
  else { o += (char)(0xF0 | (cp >> 18)); o += (char)(0x80 | ((cp >> 12) & 0x3F)); o += (char)(0x80 | ((cp >> 6) & 0x3F)); o += (char)(0x80 | (cp & 0x3F)); }
}

String jsonStr(const String& js, int i) {
  String o;
  for (; i < (int)js.length(); i++) {
    char c = js[i];
    if (c == '"') break;
    if (c != '\\') { o += c; continue; }
    char e = js[++i];
    if (e == 'n') o += '\n';
    else if (e == 'u') {
      uint32_t cp = strtoul(js.substring(i + 1, i + 5).c_str(), nullptr, 16);
      i += 4;
      if (cp >= 0xD800 && cp <= 0xDBFF && js[i + 1] == '\\' && js[i + 2] == 'u') {
        uint32_t lo = strtoul(js.substring(i + 3, i + 7).c_str(), nullptr, 16);
        cp = 0x10000 + ((cp - 0xD800) << 10) + (lo - 0xDC00);
        i += 6;
      }
      appendUtf8(o, cp);
    } else o += e;
  }
  return o;
}

long jsonNum(const String& js, const char* key, int from) {
  int p = js.indexOf(key, from);
  if (p < 0) return 0;
  return js.substring(p + strlen(key)).toInt();
}

// ---------------------- týdenní tabulka ----------------------
const char* POMLCKA = " – ";
const char* TAB_ZACATEK = "🏆 Týden";

int disciplina(uint32_t epoch) {
  time_t t = epoch; struct tm tmv;
  localtime_r(&t, &tmv);
  tmv.tm_mday -= (tmv.tm_wday + 6) % 7;
  tmv.tm_hour = 12; tmv.tm_min = 0; tmv.tm_sec = 0; tmv.tm_isdst = -1;
  time_t po = mktime(&tmv);
  long tyden = ((long)(po / 86400) + 3) / 7;
  return (int)((tyden + 2) % 3);
}

const char* discNazev(int d) {
  if (d == 0) return "🏁 0–100";
  if (d == 1) return "🏁 0–50";
  return "📏 Najeto";
}

String fmtVal(int d, float v) { return comma(String(v, 1)) + (d < 2 ? " s" : " km"); }
bool lepsi(int d, float a, float b) { return d < 2 ? a < b : a > b; }

String tabHlavicka(uint32_t epoch) {
  time_t t = epoch; struct tm tmv;
  localtime_r(&t, &tmv);
  int odPondeli = (tmv.tm_wday + 6) % 7;
  time_t po = t - odPondeli * 86400L, ne = po + 6 * 86400L;
  struct tm a, b;
  localtime_r(&po, &a);
  localtime_r(&ne, &b);
  return String(TAB_ZACATEK) + " " + String(a.tm_mday) + ". " + String(a.tm_mon + 1) + ". – " +
         String(b.tm_mday) + ". " + String(b.tm_mon + 1) + ".\n" + discNazev(disciplina(epoch));
}

float numFrom(String s) { s.replace(",", "."); s.trim(); return s.toFloat(); }

bool jeMedaile(const String& l) {
  return l.startsWith("🥇") || l.startsWith("🥈") || l.startsWith("🥉") ||
         (l.length() && isdigit((unsigned char)l[0]));
}

void lbParse(const String& text) {
  lbN = 0;
  int start = 0;
  while (start < (int)text.length() && lbN < LB_MAX) {
    int end = text.indexOf('\n', start);
    if (end < 0) end = text.length();
    String l = text.substring(start, end);
    start = end + 1;
    if (!jeMedaile(l)) continue;
    int sp = l.indexOf(' ');
    int d = l.lastIndexOf(POMLCKA);
    if (sp < 0 || d < 0 || sp >= d) continue;
    lb[lbN].label = l.substring(sp + 1, d);
    lb[lbN].val = numFrom(l.substring(d + strlen(POMLCKA)));
    if (lb[lbN].val > 0) lbN++;
  }
}

bool mujRadek(const String& label) {
  const char* ikony[] = { "🟢", "🔵", "🟣", "🟠", "🔴", "⚫" };
  for (const char* ik : ikony) if (label.startsWith(cfg.name + " " + ik)) return true;
  return false;
}

String medaile(int i) {
  if (i == 0) return "🥇";
  if (i == 1) return "🥈";
  if (i == 2) return "🥉";
  return String(i + 1) + ".";
}

String lbBuild(const String& hlavicka, int d) {
  int idx[LB_MAX];
  for (int i = 0; i < lbN; i++) idx[i] = i;
  for (int a = 0; a < lbN; a++) for (int b = a + 1; b < lbN; b++)
    if (lepsi(d, lb[idx[b]].val, lb[idx[a]].val)) { int t = idx[a]; idx[a] = idx[b]; idx[b] = t; }
  String m = hlavicka + "\n\n";
  for (int i = 0; i < lbN; i++)
    m += medaile(i) + " " + lb[idx[i]].label + POMLCKA + fmtVal(d, lb[idx[i]].val) + "\n";
  return m;
}

void najdiTabulku(long& id, String& text) {
  nactiPripnutou(id, text);
  if (text.startsWith(TAB_ZACATEK)) return;
  long mojeId = prefs.getLong("lbId", 0);
  String mujText = prefs.getString("lbTxt", "");
  if (!mojeId || !mujText.startsWith(TAB_ZACATEK)) return;
  String chyba;
  if (pripni(mojeId, chyba)) nactiPripnutou(id, text);
  else if (chyba.indexOf("not found") >= 0) {
    prefs.remove("lbId");
    prefs.remove("lbTxt");
    return;
  }
  else varovaniPripnuti(chyba);
  if (!text.startsWith(TAB_ZACATEK)) { id = mojeId; text = mujText; }
}

void updateLeaderboard(uint32_t nowEpoch, float t100, float t50, float km) {
  int d = disciplina(nowEpoch);
  float moje = d == 0 ? t100 : (d == 1 ? t50 : km);
  if (moje <= 0) return;

  String hlav = tabHlavicka(nowEpoch);
  String mujLabel = jmenoKratke(totalKm());
  delay(random(0, 4000));

  for (int pokus = 0; pokus < 3; pokus++) {
    long id; String text;
    najdiTabulku(id, text);

    lbN = 0;
    if (text.startsWith(hlav)) lbParse(text);

    int k = -1;
    for (int i = 0; i < lbN; i++) if (mujRadek(lb[i].label)) { k = i; break; }
    float nova = moje;
    if (k >= 0) {
      if (d == 2) nova = lb[k].val + moje;
      else if (!lepsi(d, moje, lb[k].val)) nova = lb[k].val;
    } else if (lbN < LB_MAX) {
      k = lbN++;
    } else {
      int nej = 0;
      for (int i = 1; i < lbN; i++) if (lepsi(d, lb[nej].val, lb[i].val)) nej = i;
      if (!lepsi(d, moje, lb[nej].val)) return;
      k = nej;
    }
    lb[k].label = mujLabel;
    lb[k].val = nova;
    String novyText = lbBuild(hlav, d);
    String mujRadekText = mujLabel + POMLCKA + fmtVal(d, nova);

    String r;
    if (id && text.startsWith(TAB_ZACATEK)) {
      int code = tgPost("editMessageText", "chat_id=" + chatId + "&message_id=" + String(id) +
                        "&text=" + urlencode(novyText), &r);
      if (code == 200 || r.indexOf("not modified") >= 0) {
        prefs.putLong("lbId", id);
        prefs.putString("lbTxt", novyText);
        delay(1500);
        long id2; String text2;
        nactiPripnutou(id2, text2);
        if (!text2.startsWith(TAB_ZACATEK) || text2.indexOf(mujRadekText) >= 0) return;
        Serial.println("Tabulku mezitim prepsal nekdo jiny, znovu");
        continue;
      }
    }

    if (tgPost("sendMessage", "chat_id=" + chatId + "&text=" + urlencode(novyText) +
               "&disable_notification=true", &r) != 200) return;
    long noveId = jsonNum(r, "\"message_id\":", 0);
    if (!noveId) return;
    prefs.putLong("lbId", noveId);
    prefs.putString("lbTxt", novyText);
    String chyba;
    if (!pripni(noveId, chyba)) varovaniPripnuti(chyba);
    return;
  }
}

  if (!http.begin(c, String(OTA_URL) + "version.txt?t=" + String(millis()))) return;
  int code = http.GET();
  String v = code == 200 ? http.getString() : String("");
  http.end();
  v.trim();
  int nova = v.toInt();
  Serial.printf("Verze: mam %d, na GitHubu %d\n", FW_VERSION, nova);
  if (code != 200 || nova <= FW_VERSION) return;

  WiFiClientSecure c2;
  c2.setInsecure();
  httpUpdate.setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS);
  httpUpdate.rebootOnUpdate(true);
  t_httpUpdate_return r = httpUpdate.update(c2, String(OTA_URL) + "firmware.bin");
  if (r == HTTP_UPDATE_FAILED)
    Serial.printf("Aktualizace selhala: %s\n", httpUpdate.getLastErrorString().c_str());
}

// ---------------------- ELM327 ----------------------
bool elmCmd(const char* cmd, String& out, uint32_t timeout) {
  while (SerialBT.available()) SerialBT.read();
  SerialBT.print(cmd);
  SerialBT.print('\r');
  out = "";
  uint32_t t0 = millis();
  while (millis() - t0 < timeout) {
    while (SerialBT.available()) {
      char c = SerialBT.read();
      if (c == '>') return true;
      if (c == ' ') continue;
      if (c == '\r' || c == '\n') {
        if (out.length() && out[out.length() - 1] != '|') out += '|';
        continue;
      }
      out += (char)toupper(c);
    }
    delay(2);
  }
  return false;
}

int hexByte(const String& s, int pos) {
  if (pos + 2 > (int)s.length()) return -1;
  char b[3] = { s[pos], s[pos + 1], 0 };
  char* e;
  long v = strtol(b, &e, 16);
  return *e ? -1 : (int)v;
}

bool queryPid(const char* cmd, const char* hdr, int nbytes, int* outBytes) {
  String r;
  if (!elmCmd(cmd, r, 1000)) { errStreak++; return false; }
  int p = r.indexOf(hdr);
  if (p < 0) { errStreak++; return false; }
  p += strlen(hdr);
  for (int i = 0; i < nbytes; i++) {
    int v = hexByte(r, p + 2 * i);
    if (v < 0) { errStreak++; return false; }
    outBytes[i] = v;
  }
  errStreak = 0;
  trip.obdOk = 1;
  return true;
}

bool elmInit() {
  String r;
  elmCmd("ATZ", r, 3000);
  delay(300);
  const char* setup[] = { "ATE0", "ATL0", "ATS0", "ATH0", "ATAT1" };
  for (const char* c : setup) if (!elmCmd(c, r, 1000)) return false;

  elmCmd("ATSP6", r, 1000);
  if (elmCmd("0100", r, 5000) && r.indexOf("4100") >= 0) return true;
  elmCmd("ATSP0", r, 1000);
  if (elmCmd("0100", r, 10000) && r.indexOf("4100") >= 0) return true;
  return false;
}

bool parseMac(const String& s, uint8_t* m) {
  unsigned int v[6];
  if (sscanf(s.c_str(), "%x:%x:%x:%x:%x:%x", &v[0], &v[1], &v[2], &v[3], &v[4], &v[5]) != 6) return false;
  for (int i = 0; i < 6; i++) {
    if (v[i] > 255) return false;
    m[i] = v[i];
  }
  return true;
}

bool btConnect() {
  obdLed = 1;
  static uint8_t macFails = 0;
  uint8_t mac[6];
  bool ok = false;

  if (parseMac(cfg.mac, mac) && cfg.mac != "01:23:45:67:89:BA" && macFails < 2) {
    ok = SerialBT.connect(mac, 0, ESP_SPP_SEC_NONE, ESP_SPP_ROLE_MASTER);
    if (ok) macFails = 0; else macFails++;
  } else {
    SerialBT.deleteAllBondedDevices();
    BTScanResults* res = SerialBT.discover(10000);
    if (res) {
      for (int i = 0; i < res->getCount(); i++) {
        BTAdvertisedDevice* d = res->getDevice(i);
        if (d && d->haveName() && String(d->getName().c_str()) == cfg.elm) {
          String a = String(d->getAddress().toString().c_str());
          a.toUpperCase();
          if (parseMac(a, mac)) {
            ok = SerialBT.connect(mac, 0, ESP_SPP_SEC_NONE, ESP_SPP_ROLE_MASTER);
            if (ok) { cfg.mac = a; saveCfgKey("mac", a); }
          }
          break;
        }
      }
    }
    if (!ok) ok = SerialBT.connect(cfg.elm);
    macFails = 0;
  }
  Serial.printf("BT pripojeni k %s: %s\n", cfg.elm.c_str(), ok ? "OK" : "selhalo");
  return ok;
}

void readVolt() {
  String r;
  if (!elmCmd("ATRV", r, 1000)) return;
  r.replace("|", "");
  float v = r.toFloat();
  if (v < 5 || v > 20) return;
  trip.battLast = (uint16_t)(v * 1000);
}

// ---------------------- výpočty ----------------------
uint32_t crossTime(uint32_t t1, int v1, uint32_t t2, int v2, int target) {
  if (v2 == v1) return t2;
  return t1 + (uint32_t)((target - v1) * (float)(t2 - t1) / (v2 - v1));
}

// Spolehlivé měření rozjezdu z 0 km/h s okamžitým uložením rekordu do paměti
void onSpeed(int v, uint32_t now) {
  if (prevSpeedMs) {
    uint32_t dt = now - prevSpeedMs;
    if (dt < 3000) {
      trip.distKm += (v + prevSpeed) / 2.0f * dt / 3600000.0f;
      if (v > 0 || prevSpeed > 0) trip.movingMs += dt;
    }
  }
  if (v > trip.maxSpeed) trip.maxSpeed = v;
  if (v > 0) lastMoveMs = now;

  if (v == 0) {
    if (prevSpeed > 0) saveNow = true;              // zastavil -> hned uložit
    if (prevSpeed > 0 || !zeroSince) zeroSince = now;
    runActive = false;                              // zastavil -> měření končí
    lastZeroMs = now;
  } else {
    zeroSince = 0;
    if (prevSpeed == 0) {                           // rozjezd z 0 -> stopky hned
      uint32_t dtZ = now - lastZeroMs;
      runStart = (lastZeroMs && dtZ < 5000) ? (lastZeroMs + now) / 2 : now;
      runActive = true;
      got50 = false;
      run050 = 0;
      run0100 = 0;
    }
    if (runActive) {
      // Po 45 s rozjezd ukončí, aby se nezablokovalo čtení otáček/teploty při plynulé jízdě pod 100 km/h
      if (now - runStart > 45000) {
        runActive = false;
      } else {
        if (!got50 && v >= 50) {
          uint32_t t0 = prevSpeedMs ? prevSpeedMs : runStart;
          uint32_t t = crossTime(t0, prevSpeed, now, v, 50) - runStart;
          if ((int32_t)t < 100) t = now - runStart;
          run050 = t;
          if (!trip.best050 || t < trip.best050) {
            trip.best050 = t;
            saveNow = true;                         // hned uložit do paměti
          }
          got50 = true;
        }
        if (v >= 100) {
          uint32_t t0 = prevSpeedMs ? prevSpeedMs : runStart;
          uint32_t t = crossTime(t0, prevSpeed, now, v, 100) - runStart;
          if ((int32_t)t < 100) t = now - runStart;
          run0100 = t;
          if (!trip.best0100 || t < trip.best0100) {
            trip.best0100 = t;
            saveNow = true;                         // hned uložit do paměti
          }
          runActive = false;
        }
      }
    }
  }
  prevSpeed = v;
  prevSpeedMs = now;
}

void periodic() {
  uint32_t now = millis();
  trip.onSec = onBase + now / 1000;

  if (portalReq) {
    saveTrip();
    prefs.putUChar("portal", 1);
    ESP.restart();
  }

  static uint32_t lastDraw = 0;
  if (now - lastDraw > 200) {
    lastDraw = now;
    drawOled();
  }

  static bool bootsReset = false;
  if (!bootsReset && now > 20000) {
    bootsReset = true;
    prefs.putUChar("boots", 0);
  }

  static uint32_t btnSince = 0;
  if (digitalRead(BTN_BOOT) == LOW) {
    if (!btnSince) btnSince = now ? now : 1;
    else if (now - btnSince > 2000) {
      saveTrip();
      prefs.putUChar("portal", 1);
      ESP.restart();
    }
  } else btnSince = 0;

  static uint32_t lastSave = 0;
  if (saveNow || now - lastSave > 15000) {
    saveNow = false;
    lastSave = now;
    saveTrip();
  }

  static uint32_t lastScan = 0;
  if (btConnected && prevSpeed == 0 && zeroSince && now - zeroSince > 3000 &&
      trip.distKm >= ARRIVE_CHECK_KM && now - lastScan > 10000) {
    lastScan = now;
    bool home = homeVisible();
    WiFi.mode(WIFI_OFF);
    if (home) {
      Serial.println("Doma -> konec jizdy, restart a odeslani");
      txLed = true;
      finishTrip();
      SerialBT.disconnect();
      delay(200);
      ESP.restart();
    }
  }
}

// ---------------------- setup / loop ----------------------
void setup() {
  Serial.begin(115200);
  oledInit();
  pinMode(LED_POWER, OUTPUT);
  pinMode(LED_OBD, OUTPUT);
  pinMode(LED_TX, OUTPUT);
  pinMode(BTN_BOOT, INPUT_PULLUP);
  pinMode(BTN_PAGE, INPUT_PULLUP);
  xTaskCreate(ledTask, "led", 2048, nullptr, 1, nullptr);
  xTaskCreate(btnTask, "btn", 2048, nullptr, 1, nullptr);

  prefs.begin("obd", false);
  loadCfg();

  bool tlacitko = prefs.getUChar("portal", 0);
  if (tlacitko) prefs.putUChar("portal", 0);
  uint8_t zapnuti = 0;
  if (esp_reset_reason() == ESP_RST_POWERON) {
    zapnuti = prefs.getUChar("boots", 0) + 1;
    prefs.putUChar("boots", zapnuti);
  }
  if (!configured() || tlacitko || zapnuti >= 3) {
    prefs.putUChar("boots", 0);
    runPortal();
  }

  introAnim();

  loadAll();

  bool home = homeVisible();
  Serial.printf("Domaci WiFi: %s\n", home ? "ANO" : "ne");

  if (home) {
    finishTrip();
    txLed = true;
    faze = 1;
    drawOled();
    bool wifiOk = wifiConnect();
    setWifiSt(wifiOk ? 2 : 1);
    if (wifiOk) {
      uint32_t nowEpoch = syncTime();
      uint8_t odeslano = 0;
      float   sT100 = 0, sT50 = 0, sKm = 0;
      while (qCount > 0) {
        if (!tgSend(tripMsg(queue[0]))) break;
        odeslano++;
        if (queue[0].best0100 && (!sT100 || queue[0].best0100 / 1000.0f < sT100)) sT100 = queue[0].best0100 / 1000.0f;
        if (queue[0].best050 && (!sT50 || queue[0].best050 / 1000.0f < sT50)) sT50 = queue[0].best050 / 1000.0f;
        sKm += queue[0].distKm;
        memmove(queue, queue + 1, sizeof(Trip) * (qCount - 1));
        qCount--;
        saveQueue();
        drawOled();
      }
      if (!nowEpoch) nowEpoch = tgDate;
      if (odeslano && nowEpoch) updateLeaderboard(nowEpoch, sT100, sT50, sKm);
      faze = qCount ? 4 : 2;
      drawOled();
      checkOta();
    } else {
      faze = 4;
      drawOled();
    }
    txLed = false;
    WiFi.disconnect(true);
    WiFi.mode(WIFI_OFF);
    startTrip();
  } else {
    WiFi.mode(WIFI_OFF);
    if (!trip.open) startTrip();
  }

  saveTrip();
  onBase = trip.onSec - millis() / 1000;
  delay(300);
  SerialBT.begin("ESP_OBD", true);
#if ESP_ARDUINO_VERSION_MAJOR >= 3
  SerialBT.setPin(cfg.pin.c_str(), cfg.pin.length());
#else
  SerialBT.setPin(cfg.pin.c_str());
#endif
}

void loop() {
  if (!btConnected) {
    static uint32_t nextTry = 0;
    if (millis() >= nextTry) {
      if (btConnect()) {
        if (elmInit()) {
          setObdSt(3);
          btConnected = true;
          errStreak = 0;
          obdLed = 2;
          faze = 0;
          stranka = 0;
          Serial.println("ELM OK");
        } else {
          setObdSt(2);
          SerialBT.disconnect();
          obdLed = 0;
        }
      } else {
        setObdSt(1);
        obdLed = 0;
      }
      nextTry = millis() + 5000;
    }
    periodic();
    delay(50);
    return;
  }

  int b[2];
  uint32_t now = millis();

  // rychlost – každou smyčku
  if (queryPid("010D", "410D", 1, b)) onSpeed(b[0], millis());

  // ostatní PIDy jen když nejedeme rozjezd, aby 0–100 měřilo přesně
  if (!runActive) {
    if (queryPid("010C", "410C", 2, b)) {
      int rpm = (b[0] * 256 + b[1]) / 4;
      liveRpm = rpm;
      if (rpm > trip.maxRpm) trip.maxRpm = rpm;
    }

    if (stranka == 1) {
      static uint32_t tMap = 0;
      if (now - tMap > 1000) {
        tMap = now;
        if (queryPid("010B", "410B", 1, b)) liveMap = b[0];
      }
      static uint32_t tLoad = 0;
      if (now - tLoad > 5000) {
        tLoad = now;
        if (queryPid("0104", "4104", 1, b)) liveLoad = (uint8_t)(b[0] * 100 / 255);
        if (queryPid("010F", "410F", 1, b)) liveIat = b[0] - 40;
      }
    }

    static uint32_t tPomal = 0;
    if (now - tPomal > 15000) {
      tPomal = now;
      if (queryPid("0105", "4105", 1, b)) {
        int c = b[0] - 40;
        liveCool = c;
        if (c > trip.maxCoolant) trip.maxCoolant = c;
      }
      readVolt();
    }

    static uint8_t dtcTries = 0;
    if (dtcTries < 3) {
      dtcTries++;
      if (queryPid("0101", "4101", 1, b)) {
        trip.mil = (b[0] & 0x80) ? 1 : 0;
        trip.dtc = b[0] & 0x7F;
        dtcTries = 3;
      }
    }
  }

  if (errStreak >= 8 || !SerialBT.connected()) {
    Serial.println("Spojeni s ELM ztraceno");
    SerialBT.disconnect();
    btConnected = false;
    obdLed = 0;
  }

  periodic();
}
